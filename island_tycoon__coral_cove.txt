let audio=null,master=null,musicTimer=null,nextNote=0,beat=0;
const melody=[76,79,83,79,74,78,81,78,72,76,79,76,74,78,81,86];
function tone(note,at,duration,gain=.06,type='sine'){
  if(!audio||!master)return;
  const o=audio.createOscillator(),g=audio.createGain();
  o.type=type;o.frequency.value=440*Math.pow(2,(note-69)/12);
  g.gain.setValueAtTime(0,at);
  g.gain.linearRampToValueAtTime(gain,at+.018);
  g.gain.exponentialRampToValueAtTime(.0001,at+duration);
  o.connect(g);g.connect(master);o.start(at);o.stop(at+duration+.03);
}
function scheduleMusic(){
  if(!audio||!musicOn||document.hidden||!started)return;
  if(nextNote<audio.currentTime)nextNote=audio.currentTime+.03;
  while(nextNote<audio.currentTime+.2){
    tone(melody[beat%melody.length],nextNote,.65,.065,'sine');
    if(beat%4===0){
      const root=[48,50,45,43][Math.floor(beat/4)%4];
      tone(root,nextNote,1.4,.055,'triangle');
      tone(root+7,nextNote,1.1,.018,'sine');
    }
    beat++;nextNote+=.32;
  }
}
function initAudio(){
  try{
    if(!audio){
      const AC=window.AudioContext||window.webkitAudioContext;
      if(!AC)return;
      audio=new AC();master=audio.createGain();master.gain.value=musicOn?.48:0;
      master.connect(audio.destination);
      musicTimer=setInterval(scheduleMusic,90);
    }
    audio.resume().catch(()=>{});
    if(musicOn)nextNote=audio.currentTime+.04;
  }catch(e){}
}
function chime(){
  if(!musicOn||!audio)return;

}
